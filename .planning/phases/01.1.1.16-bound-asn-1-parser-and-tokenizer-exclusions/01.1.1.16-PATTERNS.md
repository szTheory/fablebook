      if row["path"] == expected_path,
        do: :ok,
        else: fail!(:source_identity, {row, expected_path})

      fail!(:source_identity, {row, expected})

    :error ->

        do: fail!(:source_identity, {row, :unexpected_lane_or_path}),
        else: :ok
  end
end
