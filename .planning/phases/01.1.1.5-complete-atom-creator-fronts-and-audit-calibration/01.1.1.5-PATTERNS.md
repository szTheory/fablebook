defp compile!(files, dir, debug_info) do
  previous = Code.get_compiler_option(:debug_info)
  Code.put_compiler_option(:debug_info, debug_info)

  {result, _stderr} =
    try do
      ExUnit.CaptureIO.with_io(:stderr, fn ->
        Kernel.ParallelCompiler.compile_to_path(files, dir)
      end)

      Code.put_compiler_option(:debug_info, previous)

end

~~~elixir
def lane!(runtime) do
  name = if System.get_env("FABLEBOOK_FLOOR") in [nil, ""], do: "core", else: "floor"
  expected = %{elixir: hd(@refs[name]["elixir"]), otp: hd(@refs[name]["otp"])}

  if runtime != expected,
    do: fail!(:runtime_identity, %{lane: name, expected: expected, actual: runtime})

  Map.put(runtime, :name, name)
end
~~~

### Static source/BEAM evidence

**Source:** test/support/beam_walker.ex (tracked, reuse-only; lines 4–20) and test/contract/banned_calls_test.exs.

~~~elixir
def scan(beam_path) do
  path = to_charlist(beam_path)

  case :beam_lib.chunks(path, [:debug_info]) do
    {:ok, {_module, [debug_info: {:debug_info_v1, _backend, :none}]}} ->
      {:error, :no_debug_info}

    {:ok, {module, [debug_info: {:debug_info_v1, backend, data}]}} ->
      scan_debug_info(path, module, backend, data)

    {:error, :beam_lib, reason} ->
      {:error, reason}
  end
end
~~~

Compile fixture bodies with debug info and inspect BeamWalker.scan/1; separately assert Credo catches source forms. Preserve direct call, capture, apply, generated-per-MFA, and quote/source-only checks without executing a fixture.
